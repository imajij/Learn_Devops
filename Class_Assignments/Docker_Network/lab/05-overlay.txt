$ docker ps --filter name=k8s- --format '{{.Names}}: {{.Status}}'

### 1. turn this daemon into a one-node swarm (overlay networks need the swarm control plane)
$ docker swarm init --advertise-addr 192.168.5.1 | head -2
Swarm initialized: current node (1icef6v9prxdtgspi14iur1a1) is now a manager.

$ docker node ls
ID                            HOSTNAME   STATUS    AVAILABILITY   MANAGER STATUS   ENGINE VERSION
1icef6v9prxdtgspi14iur1a1 *   colima     Ready     Active         Leader           29.5.2
### 2. create an attachable overlay network
$ docker network create -d overlay --attachable --subnet 10.20.0.0/24 ajij-overlay

$ docker service create --quiet --name ajij-web --replicas 2 --network ajij-overlay -p 8131:80 nginx:alpine

### 5. routing mesh: the published port answers on the node's IP (the Colima VM, 192.168.5.1)
$ colima ssh -- curl -s -m 5 http://192.168.5.1:8131 | grep -o '<title>.*</title>'
<title>Welcome to nginx!</title>
### ...but not on 127.0.0.1: the ingress DNAT rule sends it to a bridge IP, which the kernel drops for loopback sources
$ colima ssh -- curl -s -m 5 -o /dev/null -w 'localhost:8131 -> HTTP %{http_code}\n' http://localhost:8131
localhost:8131 -> HTTP 000
time="2026-10-07T22:10:37+05:30" level=fatal msg="exit status 28"
$ colima ssh -- sudo iptables -t nat -S DOCKER-INGRESS
-N DOCKER-INGRESS
-A DOCKER-INGRESS -p tcp -m tcp --dport 8131 -j DNAT --to-destination 172.19.0.2:8131
-A DOCKER-INGRESS -j RETURN
### no process listens on 8131 (it is pure iptables/IPVS), so Colima has nothing to forward to the Mac
$ curl -s -m 5 http://localhost:8131 || echo 'macOS: not reachable (curl exit '$?')'
macOS: not reachable (curl exit 28)
