### 3. who is on which network
$ for n in ajij-frontend-net ajij-backend-net ajij-database-net; do echo "$n: $(docker network inspect -f '{{range .Containers}}{{.Name}}({{.IPv4Address}}) {{end}}' $n)"; done
ajij-frontend-net: ajij-backend(172.30.1.3/24) ajij-frontend(172.30.1.2/24) 
ajij-backend-net: ajij-backend(172.30.2.2/24) ajij-database(172.30.2.3/24) 
ajij-database-net: ajij-database(172.30.3.2/24)
