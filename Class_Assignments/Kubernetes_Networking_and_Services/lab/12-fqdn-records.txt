### pod A record: <pod-ip-with-dashes>.<namespace>.pod.cluster.local
$ IP=$(kubectl get pod curl -o jsonpath="{.status.podIP}"); echo $IP
10.244.0.87
$ kubectl exec dnsutils -- dig +noall +answer $(echo $IP | tr . -).default.pod.cluster.local
10-244-0-87.default.pod.cluster.local. 30 IN A	10.244.0.87
